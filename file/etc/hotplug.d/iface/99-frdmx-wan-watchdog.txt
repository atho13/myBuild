#!/bin/sh

if [ "$ACTION" = "ifup" ]; then
    
    . /lib/functions.sh
    . /lib/functions/network.sh
    
    IS_WAN_ZONE=$(ubus call network.interface dump | jsonfilter -e "@.interface[@.interface=\"$INTERFACE\"].data.zone")
    
    if [ "$IS_WAN_ZONE" = "wan" ]; then
        
        sleep 5
        
        if ! ping -c 2 -W 3 8.8.8.8 > /dev/null; then
            ubus call network.interface.$INTERFACE down
            sleep 2
            ubus call network.interface.$INTERFACE up
        fi
    fi
fi
